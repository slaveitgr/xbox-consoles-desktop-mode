cmake_minimum_required(VERSION 3.20)
project(WineExplorerSourceAdapterChecks LANGUAGES C)
enable_testing()
add_executable(wine_explorer_support_checks SupportAlgorithmsTests.c ../SupportAlgorithms.c)
target_include_directories(wine_explorer_support_checks PRIVATE ..)
set_target_properties(wine_explorer_support_checks PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
if(MSVC)
  target_compile_options(wine_explorer_support_checks PRIVATE /W4)
else()
  target_compile_options(wine_explorer_support_checks PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME wine_explorer_source_adapter_contracts COMMAND wine_explorer_support_checks)
