cmake_minimum_required(VERSION 3.20)
project(ProtonDesktopArchive LANGUAGES C CXX)
include(CTest)
set(BOXEDWINE_SOURCE "" CACHE PATH "Pristine pinned Boxedwine64 checkout")
get_filename_component(BOXEDWINE_SOURCE "${BOXEDWINE_SOURCE}" REALPATH)
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${BOXEDWINE_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE archive_pin OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT archive_pin STREQUAL "58ecb0fffdd11e05bb62e7f0e80dd1af7c6402a4")
  message(FATAL_ERROR "Archive sources must match the pinned emulator")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${BOXEDWINE_SOURCE}" status --porcelain --untracked-files=all
  OUTPUT_VARIABLE archive_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT archive_dirty STREQUAL "")
  message(FATAL_ERROR "The pinned archive source must remain pristine")
endif()
set(zlib_root "${BOXEDWINE_SOURCE}/lib/zlib")
set(zip_sources)
foreach(source adler32 compress crc32 deflate gzclose gzlib gzread gzwrite
               infback inffast inflate inftrees trees uncompr zutil)
  list(APPEND zip_sources "${zlib_root}/${source}.c")
endforeach()
foreach(source ioapi unzip zip)
  list(APPEND zip_sources "${zlib_root}/contrib/minizip/${source}.c")
endforeach()
add_library(protondesktop_zip STATIC ${zip_sources})
target_include_directories(protondesktop_zip PUBLIC
  "${zlib_root}" "${zlib_root}/contrib/minizip")
target_compile_definitions(protondesktop_zip PRIVATE _CRT_SECURE_NO_WARNINGS)
if(NOT WIN32)
  target_compile_definitions(protondesktop_zip PRIVATE _FILE_OFFSET_BITS=64 _LARGEFILE64_SOURCE)
endif()
if(BUILD_TESTING)
  add_executable(proton-archive-tests ArchiveTests.cpp)
  target_compile_features(proton-archive-tests PRIVATE cxx_std_17)
  target_link_libraries(proton-archive-tests PRIVATE protondesktop_zip)
  add_test(NAME proton-real-minizip COMMAND proton-archive-tests "${CMAKE_CURRENT_BINARY_DIR}/test-files")
endif()
